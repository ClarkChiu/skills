Common record shape every lane returns (a plain dict):
```
{lane, title, url, score:int, score_label:str, meta:str, date:str, excerpt:str, top_comment:str|None, relevance:float}
```

### T1.1 — HN lane (Algolia)

- Run → red.
- `last30d/scripts/lib/hn.py` — complete impl:
```python
"""Hacker News lane via the public Algolia Search API (no key)."""
import json, urllib.parse, urllib.request
from datetime import datetime, timezone
BASE = "https://hn.algolia.com/api/v1/search_by_date"
def search(topic, from_date, to_date, limit=25):
    frm = int(datetime.fromisoformat(from_date).replace(tzinfo=timezone.utc).timestamp())
    q = urllib.parse.urlencode({"query": topic, "tags": "story",
        "numericFilters": f"created_at_i>{frm}", "hitsPerPage": max(limit*2, 50)})
    try:
        with urllib.request.urlopen(f"{BASE}?{q}", timeout=15) as r:
            hits = json.loads(r.read().decode()).get("hits", [])
    except Exception:
        return []
    out = []
    for h in hits:
        pts = int(h.get("points") or 0)
        out.append({"lane": "hn", "title": (h.get("title") or "").strip(),
            "url": f"https://news.ycombinator.com/item?id={h.get('objectID')}",
            "score": pts, "score_label": "points",
            "meta": f"{h.get('num_comments') or 0} comments",
            "date": (h.get("created_at") or "")[:10],
            "excerpt": (h.get("story_text") or "")[:200], "top_comment": None,
            "relevance": 0.0})
    out.sort(key=lambda x: x["score"], reverse=True)
    return out[:limit]
```
- Run → green. **Commit**: `✨ feat(last30d): HN lane (Algolia API)`.

### T1.2 — arXiv lane (Atom API)
- `last30d/tests/test_arxiv.py`: feed a canned arXiv Atom XML (2 entries), assert 2 records, `lane=="arxiv"`, `score_label=="arxiv"`, url is the abs link, date parsed, sorted by date desc (arXiv has no engagement score → rank by recency).
- Run → red.
- `last30d/scripts/lib/arxiv.py` — complete impl:
```python
"""arXiv lane via the public arXiv API (Atom, no key). No engagement metric → rank by recency."""
import urllib.parse, urllib.request
import xml.etree.ElementTree as ET
ATOM = "{http://www.w3.org/2005/Atom}"
BASE = "http://export.arxiv.org/api/query"
def search(topic, from_date, to_date, limit=25):
    q = urllib.parse.urlencode({"search_query": f"all:{topic}", "sortBy": "submittedDate",
        "sortOrder": "descending", "max_results": max(limit*2, 40)})
    try:
        with urllib.request.urlopen(f"{BASE}?{q}", timeout=20) as r:
            root = ET.fromstring(r.read().decode())
    except Exception:
        return []
    out = []
    for e in root.iter(f"{ATOM}entry"):
        pub = (e.findtext(f"{ATOM}published") or "")[:10]
        if pub and (pub < from_date or pub > to_date):
            continue
        title = " ".join((e.findtext(f"{ATOM}title") or "").split())
        link = ""
        for l in e.iter(f"{ATOM}link"):
            if l.get("rel") == "alternate":
                link = l.get("href", "")
        authors = [a.findtext(f"{ATOM}name") or "" for a in e.iter(f"{ATOM}author")]
        out.append({"lane": "arxiv", "title": title, "url": link, "score": 0,
            "score_label": "arxiv", "meta": ", ".join(authors[:3]), "date": pub,
            "excerpt": " ".join((e.findtext(f"{ATOM}summary") or "").split())[:200],
            "top_comment": None, "relevance": 0.0})
    out.sort(key=lambda x: x["date"], reverse=True)
    return out[:limit]
```
- Run → green. **Commit**: `✨ feat(last30d): arXiv lane (arXiv API)`.

### T1.3 — GitHub lane (REST Search)
- `last30d/tests/test_github.py`: monkeypatch urllib to canned repo-search JSON (2 items with `stargazers_count`, `full_name`, `html_url`, `pushed_at`, `description`); assert 2 records, `lane=="github"`, `score_label=="stars"`, sorted by stars, uses `GITHUB_TOKEN` header if present.

- `last30d/scripts/lib/github.py` — complete impl (repos recently pushed matching topic; `Authorization` header when `GITHUB_TOKEN` set):
```python
"""GitHub lane via the REST Search API. Uses GITHUB_TOKEN if present (higher rate limit)."""
import json, os, urllib.parse, urllib.request
BASE = "https://api.github.com/search/repositories"
def search(topic, from_date, to_date, limit=25):
    q = f"{topic} pushed:>={from_date}"
    url = f"{BASE}?" + urllib.parse.urlencode({"q": q, "sort": "stars", "order": "desc",
        "per_page": min(max(limit, 10), 50)})
    hdr = {"Accept": "application/vnd.github+json", "User-Agent": "last30d"}
    tok = os.environ.get("GITHUB_TOKEN")
    if tok:
        hdr["Authorization"] = f"Bearer {tok}"
    try:
        req = urllib.request.Request(url, headers=hdr)
        with urllib.request.urlopen(req, timeout=15) as r:
            items = json.loads(r.read().decode()).get("items", [])
    except Exception:
        return []
    out = []
    for it in items:
        stars = int(it.get("stargazers_count") or 0)
        out.append({"lane": "github", "title": it.get("full_name") or "",
            "url": it.get("html_url") or "", "score": stars, "score_label": "stars",
            "meta": (it.get("language") or ""), "date": (it.get("pushed_at") or "")[:10],
            "excerpt": (it.get("description") or "")[:200], "top_comment": None,
            "relevance": 0.0})
    out.sort(key=lambda x: x["score"], reverse=True)
    return out[:limit]
```
- Run → green. **Commit**: `✨ feat(last30d): GitHub lane (REST Search)`.

### T1.4 — YouTube lane (optional, yt-dlp)
- `last30d/tests/test_youtube.py`: monkeypatch `subprocess.run` to return canned yt-dlp JSON lines (2 videos w/ `view_count`,`like_count`,`title`,`webpage_url`,`upload_date`); assert 2 records `lane=="youtube"`, `score_label=="views"`; AND when `shutil.which("yt-dlp")` is None → `search()` returns `[]` and sets a skip reason (Rule 12 graceful degrade).
- Run → red.
- `last30d/scripts/lib/youtube.py` — complete impl:
```python
"""YouTube lane via yt-dlp (optional external CLI). Absent yt-dlp → [] + skip reason."""
import json, shutil, subprocess
SKIP = {"reason": None}
def available():
    return shutil.which("yt-dlp") is not None
def search(topic, from_date, to_date, limit=25):
    if not available():
        SKIP["reason"] = "yt-dlp not installed"
        return []
    try:
        p = subprocess.run(["yt-dlp", f"ytsearch{max(limit,10)}:{topic}",
            "--dump-json", "--flat-playlist", "--no-warnings"],
            capture_output=True, text=True, timeout=90)
    except Exception:
        SKIP["reason"] = "yt-dlp failed"
        return []
    out = []
    for line in p.stdout.splitlines():
        try:
            v = json.loads(line)
        except ValueError:
            continue
        up = v.get("upload_date") or ""
        d = f"{up[:4]}-{up[4:6]}-{up[6:8]}" if len(up) == 8 else ""
        if d and (d < from_date or d > to_date):
            continue
        views = int(v.get("view_count") or 0)
        out.append({"lane": "youtube", "title": v.get("title") or "",
            "url": v.get("webpage_url") or v.get("url") or "", "score": views,
            "score_label": "views", "meta": f"{v.get('like_count') or 0} likes",
            "date": d, "excerpt": (v.get("description") or "")[:200],
            "top_comment": None, "relevance": 0.0})
    out.sort(key=lambda x: x["score"], reverse=True)
    return out[:limit]
```
- Run → green. **Commit**: `✨ feat(last30d): YouTube lane (yt-dlp, optional)`.
